Before (Hacky Workaround)

// Had to call detect_dialect_from_text with dummy data

const wordlistSizes = new Map();

function populateWordlistSizes() {
  const sampleText = "abandon ability able about above";
  const matches = JSON.parse(detect_dialect_from_text(sampleText));

  matches.forEach(match => {
    const key = `${match.language}/${match.wordlist}`;
    wordlistSizes.set(key, match.wordlist_size);
  });
}

function getWordlistSize(language, wordlist) {
  const key = `${language}/${wordlist}`;
  return wordlistSizes.get(key) || 2048; // Fallback
}

function calculatePayloadWordBits(wordCount, wordlistSize) {
  return Math.floor(wordCount * Math.log2(wordlistSize));
}

After (Direct API)

// Clean, direct function calls

import { get_wordlist_size, get_bits_per_word } from './glossia.js';

function calculatePayloadWordBits(wordCount, language, wordlist) {
  const result = JSON.parse(get_bits_per_word(language, wordlist));

  if (result.error) {
    console.error('Invalid wordlist:', result.error);
    return 0;
  }

  return wordCount * result.bits_per_word;
}

// Even simpler - just get the size directly
function getWordlistSize(language, wordlist) {
  const result = JSON.parse(get_wordlist_size(language, wordlist));
  return result.size || 0;
}

Benefits

  • ✅ Semantically clear — function name matches intent
  • ✅ No dummy data — no need for sample text
  • ✅ No global cache — no Map to maintain
  • ✅ Type-safe errors — JSON error responses
  • ✅ More efficient — O(1) lookup from compile-time index
  • ✅ Correct by construction — uses exact precomputed sizes

API Reference

get_wordlist_size(language: string, wordlist: string): string
Returns: { "size": 2048 } or { "error": "Unknown wordlist: ..." }

get_bits_per_word(language: string, wordlist: string): string
Returns: { "bits_per_word": 11 } or { "error": "..." }

Note: Returns error if wordlist size is not a power of two
      (required for efficient encoding without padding ambiguity)
